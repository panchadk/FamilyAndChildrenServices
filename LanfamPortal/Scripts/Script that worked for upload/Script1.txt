/* ============================================================
   LANFAM legacy system -> SQL Server import  (plain SSMS version)
   Creates all tables and imports the 19 extracted CSV files,
   with relationships (FKs), indexes, and integrity reporting.

   BEFORE RUNNING (2 steps):
   1. Unzip all 19 CSV files to  C:\LANFAM\csv  on the SQL Server
      machine (the path is read by the SQL Server service, not by
      SSMS - if SSMS runs on your PC but SQL Server is remote, the
      files must be on the server or a UNC path its service
      account can read).
   2. If you use a different folder, Ctrl+H replace
      C:\LANFAM\csv  with your path before executing.

   Then just press F5. No SQLCMD mode needed.
   Creates database [LanfamArchive] if absent (Ctrl+H to rename).
   Requires SQL Server 2017 or later (BULK INSERT FORMAT='CSV').

   DESIGN NOTES
   - Each CSV loads into a staging table (all NVARCHAR), then into
     a typed table via TRY_CONVERT, so a bad value can never abort
     the load; conversion failures land as NULL.
   - Foreign keys are created WITH NOCHECK: they document and
     enforce relationships for NEW rows without rejecting legacy
     orphans (e.g. history rows for children absent from the 2008
     CHILD snapshot). The orphan report at the end quantifies them.
   - FamilyIndex consolidates the two family masters (numeric and
     S/J-prefixed keys) into one referenceable parent.
   - ChildHistory gets a persisted computed column ResnoFull
     (res_type + resno) to join Resource.
   ============================================================ */

IF DB_ID('LanfamArchive') IS NULL
    CREATE DATABASE [LanfamArchive];
GO
USE [LanfamArchive];
GO
IF SCHEMA_ID('lanfam') IS NULL
    EXEC('CREATE SCHEMA lanfam');

    CodesID INT IDENTITY(1,1) NOT NULL CONSTRAINT PK_Codes PRIMARY KEY,
    Source VARCHAR(10) NOT NULL,
    category NVARCHAR(10) NULL,
    code NVARCHAR(10) NULL,
    description NVARCHAR(60) NULL
);
INSERT INTO lanfam.Codes (Source, category, code, description)
SELECT 'CASE', category, code, description FROM lanfam.CodesCase
UNION
SELECT 'FAMILY', category, code, description FROM lanfam.CodesFamily;
CREATE INDEX IX_Codes_Lookup ON lanfam.Codes (category, code) INCLUDE (description);
GO

/* ---------- Consolidated family key parent ---------- */
IF OBJECT_ID('lanfam.FamilyIndex') IS NOT NULL DROP TABLE lanfam.FamilyIndex;
CREATE TABLE lanfam.FamilyIndex (
    famno NVARCHAR(20) NOT NULL CONSTRAINT PK_FamilyIndex PRIMARY KEY,
    SourceTable VARCHAR(10) NOT NULL
);
INSERT INTO lanfam.FamilyIndex (famno, SourceTable)
SELECT famno, MIN(t) FROM (
    SELECT famno, 'FAMILY' AS t FROM lanfam.Family WHERE famno IS NOT NULL
    UNION ALL
    SELECT famno, 'FAM2' FROM lanfam.Fam2 WHERE famno IS NOT NULL
) u GROUP BY famno;
GO

/* ---------- Reporting view: history decoded ---------- */
CREATE OR ALTER VIEW lanfam.vChildPlacementHistory AS
SELECT  h.caseno, c.surname, c.givename, c.birthdate,
        h.event_date, h.event_code, h.event_desc,
        h.ResnoFull AS resno, r.surname1 AS res_surname,
        r.category AS res_category, h.rate1, h.rate2,
        h.to_date, h.svc_code, h.svc_desc, h.comment
FROM lanfam.ChildHistory h
LEFT JOIN lanfam.Child    c ON c.caseno = h.caseno
LEFT JOIN lanfam.Resource r ON r.resno = h.ResnoFull;
GO

/* ---------- Load + integrity report ---------- */
SELECT t.name AS TableName, SUM(p.rows) AS Rows
FROM sys.tables t JOIN sys.partitions p
  ON p.object_id = t.object_id AND p.index_id IN (0,1)
WHERE SCHEMA_NAME(t.schema_id) = 'lanfam'
GROUP BY t.name ORDER BY t.name;

SELECT 'ChildHistory orphan caseno' AS Issue, COUNT(*) AS Rows
FROM lanfam.ChildHistory h
WHERE h.caseno IS NOT NULL
  AND NOT EXISTS (SELECT 1 FROM lanfam.Child c WHERE c.caseno = h.caseno)
UNION ALL
SELECT 'CostPurchase orphan caseno', COUNT(*)
FROM lanfam.CostPurchase p
WHERE p.caseno IS NOT NULL
  AND NOT EXISTS (SELECT 1 FROM lanfam.Child c WHERE c.caseno = p.caseno)
UNION ALL
SELECT 'Referral famno not in masters', COUNT(*)
FROM lanfam.Referral r
WHERE r.famno IS NOT NULL
  AND NOT EXISTS (SELECT 1 FROM lanfam.FamilyIndex f WHERE f.famno = r.famno)
UNION ALL
SELECT 'ChildHistory resno not in Resource', COUNT(*)
FROM lanfam.ChildHistory h
WHERE h.ResnoFull <> ''
  AND NOT EXISTS (SELECT 1 FROM lanfam.Resource r WHERE r.resno = h.ResnoFull);
